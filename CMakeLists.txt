cmake_minimum_required(VERSION 3.22)
project(RJM LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WITH_PIC ON CACHE BOOL "" FORCE)
if(ANDROID)
    set(PLATFORM Android CACHE STRING "" FORCE)
endif()
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git
    GIT_TAG c1ab645ca298a2801097931d1079b10ff7eb9df8)
FetchContent_GetProperties(raylib)
if(NOT raylib_POPULATED)
    FetchContent_Populate(raylib)
endif()
if(ANDROID)
    # Hook the NDK input queue, rather than polling pointers once per frame.
    # This preserves short taps, multi-touch IDs, UP and CANCEL events.
    set(android_source "${raylib_SOURCE_DIR}/src/platforms/rcore_android.c")
    file(READ "${android_source}" platform_source)
    if(NOT platform_source MATCHES "rjm_android_input")
        set(anchor "static int32_t AndroidInputCallback(struct android_app *app, AInputEvent *event)\n{")
        string(FIND "${platform_source}" "${anchor}" anchor_position)
        if(anchor_position EQUAL -1)
            message(FATAL_ERROR "Pinned raylib Android input hook changed")
        endif()
        string(REPLACE "${anchor}" "extern int rjm_android_input(struct android_app *app, AInputEvent *event);\n${anchor}\n    if (rjm_android_input(app, event)) return 1;" platform_source "${platform_source}")
        file(WRITE "${android_source}" "${platform_source}")
    endif()
endif()
add_subdirectory("${raylib_SOURCE_DIR}" "${raylib_BINARY_DIR}")
file(GLOB_RECURSE game_sources CONFIGURE_DEPENDS game/src/*.cpp)
if(ANDROID)
    add_library(rjm SHARED ${game_sources})
    target_include_directories(rjm PRIVATE "${ANDROID_NDK}/sources/android/native_app_glue")
    target_link_options(rjm PRIVATE "-Wl,-u,ANativeActivity_onCreate" "-Wl,-z,max-page-size=16384")
else()
    add_executable(rjm ${game_sources})
endif()
target_include_directories(rjm PRIVATE game/include)
target_link_libraries(rjm PRIVATE raylib)
target_compile_options(rjm PRIVATE -Wall -Wextra -Wpedantic)
